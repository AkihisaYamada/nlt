\begin{definition}[minimal inference rules]
\[
	\infer{\psi}{\phi & \phi \imp \psi}
\]
\[
	\infer[\text{ if }x \notin \Var(\phi_1,\dots,\phi_n)]{
		\phi_1 \imp\dots\imp \phi_n \imp \forall x.\ \psi
	}{
		\phi_1 \imp\dots\imp \phi_n \imp \psi
	}
\]
\[
	\infer{\phi(x\mapsto t)\imp\psi}{(\forall x.\ \phi)\imp\psi}
\]
\end{definition}

\begin{definition}[axioms]
\[
\forall \phi.\ \phi \imp \phi
\]
\[
\forall \phi,\psi,\tau.\ (\phi \imp \psi \imp \tau) \imp (\psi \imp \phi \imp \tau)
\]
\[
\forall \phi,\psi.\ \phi \imp \psi \imp \phi
\]
\[
\forall \phi,\psi.\ (\phi \imp \phi \imp \psi) \imp (\phi \imp \psi)
\]
\end{definition}

\begin{example}
\[
	\infer{a \imp (a\imp b) \imp b}{
		\infer{a \imp (a\imp b)\imp a}{
			\infer{a \imp a}{\forall \phi.\ \phi \imp \phi}
		} &
		\infer{(a \imp (a\imp b)\imp a) \imp a \imp (a\imp b) \imp b}{
			\infer{a \imp (a\imp b) \imp b}{
				\infer{(a\imp b)\imp a \imp b}{\forall \phi.\ \phi \imp \phi}
			}
		}
	}
\]
\end{example}

\begin{definition}[equality]
\[
	\forall x.\ x = x
\]
\[
	\forall x,y.\ x = y \imp y = x
\]
\[
	\forall x,y,z.\ x = y \imp y = z \imp x = y
\]
\[
	\forall f,g,x,y.\ f = g \imp x = y \imp f\,x = g\,y
\]
\end{definition}

\begin{definition}[logic symbols]
\begin{align*}
	\true &: \defined&
	\true &= (\forall \phi.\ \phi \imp \phi)\\
	\false &: \defined&
	\false &= (\forall \phi.\ \phi)
\end{align*}
\[
\forall \phi,\psi.\ (\phi \land \psi) = (\forall \rho.\ (\phi \imp \psi \imp \rho) \imp \rho)
\]
\[
\forall \phi,\psi.\ (\phi \lor \psi) = (\forall \rho.\ (\phi \imp \rho) \imp (\psi \imp \rho) \imp \rho)
\]
\[
\forall \phi.\ \neg\phi = (\phi \imp \false)
\]
\[
\forall p.\ (\exists) p = (\forall\phi.\ (\forall x.\ p\,x \imp \phi) \imp \phi)
\]
\[	\forall f,\sigma,\tau.\ f : \sigma \to \tau \defeq \forall x : \sigma.\ f\,x : \tau
\]
\[	\forall f,\sigma,\tau.\ f \in \sigma \to \tau \defeq \forall x \in \sigma.\ f\,x \in \tau
\]
\end{definition}

\begin{lemma}
\[
\infer{\forall p,x.\ p\,x \imp (\exists)p}{
	\infer{p\,x \imp (\exists)p}{
		\infer{p\,x \imp \forall \phi.\ (\forall x.\ p\,x \imp \phi) \imp \phi}{
			\infer{p\,x \imp (\forall x.\ p\,x \imp \phi) \imp \phi}{
				\infer{(\forall x.\ p\,x \imp \phi) \imp p\,x \imp \phi}{
					(p\,x\imp\phi) \imp p\,x \imp \phi
				}
			}
		}
	}
}
\]
\end{lemma}
