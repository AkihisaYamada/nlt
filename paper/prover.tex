\[ s,t,u,\ldots ::= x \mid s\,s \mid x.\,s \mid x[s]

Substitution
\begin{align*}
[x\mapsto s]x &= s\\
[x\mapsto s]y &= y \\
[x\mapsto s](t\,u) &= [x\mapsto s]t\,[x\mapsto s]u\\
[x\mapsto s](x.\,t) &= x.\,t\\
[x\mapsto s](y.\,t) &= y.\,[x\mapsto s]t\\
[x\mapsto y](x[t]) &= y[[x\mapsto y]t]\\
[x\mapsto y.s](x[t]) &= [y\mapsto t]s\\
[x\mapsto s](y[t]) &= y[[x\mapsto s]t]
\end{align*}


\subsection{Proof Contexts}

We assume an infix symbol $\IMP$ and a prefix $\forall$.

\begin{definition}[proof contexts]\label{def:ctxt}
A \emph{proof context} $\Gamma \in \Ctxt(V)$ over variables $V$ is defined as follows.

\begin{itemize}
\item $\epsilon \in \Ctxt(\set{\IMP,\forall})$
\item $(\Gamma;\ASS s) \in \Ctxt(V)$ if $\Gamma \in \Ctxt(V)$ and $t \in \Term(V)$
\item $(\Gamma;\FIX x) \in \Ctxt(V)$ if $x \in V$ and $\Gamma \in \Ctxt(V\setminus\set{x})$
\end{itemize}

\noindent A term $s$ is a \emph{theorem} in $\Gamma$, written $\Gamma\vdash s$, if it is derivable by the following inference rules:
%
\begin{align*}
\infer[\ass]{\Gamma;\ASS s\vdash s}{}
&&
\infer[\text{ if }\Gamma \in \Ctxt(V) \land t \in \Term(V)]{\Gamma;\alpha \vdash t}{\Gamma \vdash t}
\end{align*}
\begin{align*}
\infer[\impE]{\Gamma\vdash t}{\Gamma\vdash s &\Gamma\vdash s \IMP t}
&&
\infer[\impI]{\Gamma\vdash s \IMP t}{\Gamma;\ASS s\vdash t}
\end{align*}
\begin{align*}
\infer[\allE]{\Gamma\vdash s[x\mapsto t]}{\Gamma\vdash \ALL{x}{s}}
&&
\infer[\allI]{\Gamma\vdash \ALL{x}{s}}{\Gamma;\FIX x \vdash s}
\end{align*}
\end{definition}

\subsection{Specification}

\begin{definition}[specifications]\label{def:spec}
We extend \prettyref{def:ctxt} with the following case:

\begin{itemize}
\item $(\Gamma;\SPEC{x}{s}) \in \Ctxt(V)$ if $x \in V$, $s \in \Term(V)$, $\Gamma \in \Ctxt(V\setminus\set{x})$, and $\Gamma \vdash \exists x.\,s$
\end{itemize}

\noindent and the following inference rule:
\[
\infer[\spec]{\Gamma;\SPEC{x}{s}\vdash s}{}
\]

\end{definition}

By stating $\SPEC{x}{s}$ (typically $x$ occurs in $s$), one obtains a term $x$ that satisfies $s$.
To this end, one must prove that such a term exists ($\Gamma \vdash \exists x.\,s$).

\begin{metatheorem}[spec elimination]
Let $(\Gamma;\SPEC{x}{s}) \in \Ctxt(V)$ and $t \in \Term(V\setminus\set{x})$
such that $\Gamma;\SPEC{x}{s} \vdash t$.
Then $\Gamma \vdash t$.
\end{metatheorem}
\begin{proof}
It is easy to see that $\Gamma;\FIX x;\ASS s \vdash t$ by induction on the proof tree.
Thus we have $\Gamma \vdash \ALL{x}{s\IMP t}$.
On the other hand, by \prettyref{def:spec} we have $\Gamma \vdash \exists x.\,s$, i.e., $\Gamma \vdash \ALL{P}{(\ALL x{s \IMP P}) \IMP P}$. We conclude as follows:
\begin{equation}
\infer[\impE]{\Gamma \vdash t}{\Gamma \vdash \ALL x{s\IMP t} &
	\infer[\allE]{\Gamma \vdash (\ALL x{s\IMP t}) \IMP t}{
		\Gamma \vdash \ALL P{(\ALL x{s \IMP P}) \IMP P}
	}
}\tag*{\qedhere}
\end{equation}
\end{proof}

\subsection{Equality}

The root context starts with fixing the $=$ symbol:
\[
\FIX (=);
\]
Then it assumes two axioms on it:
\begin{align*}
&\ASS[EQ.refl] \ALL x{x = x};\\
&\ASS[EQ.mono] \ALL x \ALL y x = y \IMP \ALL C{C[x]\IMP C[y]};
\end{align*}

\begin{metatheorem}[definition]
$(\Gamma;\SPEC x{x = s}) \in \Ctxt(V)$ if $x\in V$, $s \in \Term(V\setminus\set{x})$, and $\Gamma \in \Ctxt(V\setminus\set{x})$.
\end{metatheorem}
